def max_drawdown(nav_series):

    cumulative = (
        nav_series
        /
        nav_series.iloc[0]
    )

    running_max = cumulative.cummax()

    drawdown = (
        cumulative
        -
        running_max
    ) / running_max

    return drawdown.min()

def calculate_var_cvar(x):

    var95 = np.percentile(x.dropna(),5)

    cvar95 = (
        x[x<=var95]
        .mean()
    )

    return pd.Series({
        'VaR_95':var95,
        'CVaR_95':cvar95
    })

  1. Risk-adjusted performance varied significantly across schemes, with Sharpe Ratios indicating substantial differences in return generated per unit of risk.
